// plan: points are tenant-scoped and never negative
// write: sealed Result, one ledger per tenant
// test: earn then redeem leaves the right balance

sealed class Result<T> {
  const Result();
}

final class Ok<T> extends Result<T> {
  const Ok(this.value);
  final T value;
}

final class Fail<T> extends Result<T> {
  const Fail(this.reason);
  final String reason;
}

class LoyaltyRepository {
  LoyaltyRepository(this._ledger);
  final Map<String, int> _ledger;

  Result<int> redeem(String tenantId, int points) {
    final balance = _ledger[tenantId] ?? 0;
    if (points <= 0 || points > balance) {
      return const Fail('insufficient points');
    }
    _ledger[tenantId] = balance - points;
    return Ok(_ledger[tenantId]!);
  }
}
